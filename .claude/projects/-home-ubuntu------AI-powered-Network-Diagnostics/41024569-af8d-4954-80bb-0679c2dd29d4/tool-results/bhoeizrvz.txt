GrpcClient::GrpcClient(const std::string& server_address) {
    // 创建gRPC通道
    channel_ = grpc::CreateChannel(server_address, grpc::InsecureChannelCredentials());
    
    // 创建服务存根
    stub_ = monitor::proto::QueryService::NewStub(channel_);
}

GrpcClient::~GrpcClient() {
}

bool GrpcClient::GetLatestScore(std::vector<ServerScoreSummary>& servers, ClusterStats& stats) {
    monitor::proto::QueryLatestScoreRequest request;
    monitor::proto::QueryLatestScoreResponse response;
    grpc::ClientContext context;

    // 调用RPC
    grpc::Status status = stub_->QueryLatestScore(&context, request, &response);
    if (!status.ok()) {
        qDebug() << "GetLatestScore RPC failed: " << status.error_message().c_str();
        return false;
    }

    // 转换服务器评分数据
    for (const auto& server : response.servers()) {
        ServerScoreSummary summary;
        summary.server_name = server.server_name();
        summary.score = server.score();
        summary.last_update = TimeUtils::TimestampToQDateTime(server.last_update());
        summary.online = (server.status() == monitor::proto::ONLINE);
        summary.cpu_percent = server.cpu_percent();
        summary.mem_used_percent = server.mem_used_percent();
        summary.disk_util_percent = server.disk_util_percent();
        summary.load_avg_1 = server.load_avg_1();
        servers.push_back(summary);
    }

    // 转换集群统计数据
    const auto& cluster_stats = response.cluster_stats();
    stats.total_servers = cluster_stats.total_servers();
    stats.online_servers = cluster_stats.online_servers();
    stats.offline_servers = cluster_stats.offline_servers();
    stats.avg_score = cluster_stats.avg_score();
    stats.max_score = cluster_stats.max_score();
    stats.min_score = cluster_stats.min_score();
    stats.best_server = cluster_stats.best_server();
    stats.worst_server = cluster_stats.worst_server();

    return true;
}

#include "data_fetch_thread.h"
#include "grpc_client.h"

#include <QDateTime>
#include <QDebug>

DataFetchThread::DataFetchThread(GrpcClient* client, QObject* parent)
    : QThread(parent), client_(client), running_(true), update_interval_(5) {
}

DataFetchThread::~DataFetchThread() {
    Stop();
    wait();
}

void DataFetchThread::SetServerName(const QString& server_name) {
    QMutexLocker locker(&mutex_);
    server_name_ = server_name;
}

void DataFetchThread::Stop() {
    running_ = false;
}

void DataFetchThread::SetUpdateInterval(int seconds) {
    update_interval_ = seconds;
}

void DataFetchThread::run() {
    while (running_) {
        // 获取最新评分（每次循环都获取）
        FetchLatestScore();
        
        // 获取当前选中服务器的详细数据
        QString server_name;
        {
            QMutexLocker locker(&mutex_);
            server_name = server_name_;
        }
        
        if (!server_name.isEmpty()) {
            FetchPerformanceData();
            FetchTrendData();
            FetchAnomalyData();
            FetchDetailData();
        }
        
        // 等待指定的更新间隔
        sleep(update_interval_);
    }
}

void DataFetchThread::FetchLatestScore() {
    std::vector<ServerScoreSummary> servers;
    ClusterStats stats;
    
    if (client_->GetLatestScore(servers, stats)) {
        emit latestScoreFetched(servers, stats);
    }
}

void DataFetchThread::FetchPerformanceData() {
    QString server_name;
    {
        QMutexLocker locker(&mutex_);
        server_name = server_name_;
    }
    
    if (server_name.isEmpty()) return;
    
    // 获取最近1小时的性能数据
    QDateTime end_time = QDateTime::currentDateTime();
    QDateTime start_time = end_time.addSecs(-3600);
    
    std::vector<PerformanceRecord> records;
    if (client_->QueryPerformance(server_name.toStdString(), start_time, end_time, records)) {
        emit performanceDataFetched(records);
    }
}

void DataFetchThread::FetchTrendData() {
    QString server_name;
    {
        QMutexLocker locker(&mutex_);
        server_name = server_name_;
    }
    
    if (server_name.isEmpty()) return;
    
    // 获取最近24小时的趋势数据，每小时聚合一次
    QDateTime end_time = QDateTime::currentDateTime();
    QDateTime start_time = end_time.addDays(-1);
    
    std::vector<PerformanceRecord> records;
    if (client_->QueryTrend(server_name.toStdString(), start_time, end_time, 3600, records)) {
        emit trendDataFetched(records);
    }
}

void DataFetchThread::FetchAnomalyData() {
    QString server_name;
    {
        QMutexLocker locker(&mutex_);
        server_name = server_name_;
    }
    
    if (server_name.isEmpty()) return;
    
    // 获取最近24小时的异常数据
    QDateTime end_time = QDateTime::currentDateTime();
    QDateTime start_time = end_time.addDays(-1);
    
    std::vector<AnomalyRecord> records;
    if (client_->QueryAnomaly(server_name.toStdString(), start_time, end_time, records)) {
        emit anomalyDataFetched(records);
    }
}

void DataFetchThread::FetchDetailData() {
    QString server_name;
    {
        QMutexLocker locker(&mutex_);
        server_name = server_name_;
    }
    
    if (server_name.isEmpty()) return;
    
    // 获取最近1小时的详细数据
    QDateTime end_time = QDateTime::currentDateTime();
    QDateTime start_time = end_time.addSecs(-3600);
    
    // 网络详细数据
    std::vector<NetDetailRecord> net_records;
    if (client_->QueryNetDetail(server_name.toStdString(), start_time, end_time, net_records)) {
        emit netDetailFetched(net_records);
    }
    
    // 磁盘详细数据
    std::vector<DiskDetailRecord> disk_records;
    if (client_->QueryDiskDetail(server_name.toStdString(), start_time, end_time, disk_records)) {
        emit diskDetailFetched(disk_records);
    }
    
    // 内存详细数据
    std::vector<MemDetailRecord> mem_records;
    if (client_->QueryMemDetail(server_name.toStdString(), start_time, end_time, mem_records)) {
        emit memDetailFetched(mem_records);
    }
    
    // 软中断详细数据
    std::vector<SoftIrqDetailRecord> softirq_records;
    if (client_->QuerySoftIrqDetail(server_name.toStdString(), start_time, end_time, softirq_records)) {
        emit softIrqDetailFetched(softirq_records);
    }
}

        static ThemeManager instance;
        return &instance;
    }

    void SetTheme(Theme theme) {
        current_theme_ = theme;
        ApplyTheme();
        SaveSettings();
        emit ThemeChanged(theme);
    }

    Theme GetTheme() const {
        return current_theme_;
    }

    QString GetThemeName() const {
        switch (current_theme_) {
            case Dark:   return tr("深色主题");
            case Light:  return tr("浅色主题");
            case Classic: return tr("经典主题");
            default:     return tr("未知");
        }
    }

    void LoadSettings() {
        QSettings settings("MonitorSystem", "QtClient");
        int theme = settings.value("theme", static_cast<int>(Dark)).toInt();
        current_theme_ = static_cast<Theme>(theme);
        ApplyTheme();
    }

signals:
    void ThemeChanged(Theme theme);

private:
    ThemeManager() : current_theme_(Dark) {}
    ThemeManager(const ThemeManager&) = delete;
    ThemeManager& operator=(const ThemeManager&) = delete;

    void ApplyTheme() {
        QString qss_file;
        
        switch (current_theme_) {
            case Dark:
                qss_file = ":/styles/dark_theme.qss";
                break;
            case Light:
                qss_file = ":/styles/light_theme.qss";
                break;
            case Classic:
                qss_file = ":/styles/main.qss";
                break;
        }

        QFile file(qss_file);
        if (file.open(QFile::ReadOnly | QFile::Text)) {
            QString style_sheet = QString::fromUtf8(file.readAll());
            qApp->setStyleSheet(style_sheet);
            file.close();
            qDebug() << "Theme applied:" << GetThemeName();
        } else {
            qWarning() << "Failed to load theme file:" << qss_file;
        }
    }

    void SaveSettings() {
        QSettings settings("MonitorSystem", "QtClient");
        settings.setValue("theme", static_cast<int>(current_theme_));
    }

    Theme current_theme_;
};
