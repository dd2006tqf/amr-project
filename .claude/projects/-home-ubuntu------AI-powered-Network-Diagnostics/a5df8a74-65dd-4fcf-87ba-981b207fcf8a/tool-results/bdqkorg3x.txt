+/// 三元组连接身份主键：控制器 + 12位句柄 + 连接代次（防御句柄复用串设备）
+struct HciConnectionKey {
+    std::string controller_id;
+    uint16_t handle{0};
+    uint32_t generation{0};
+
+    bool operator==(const HciConnectionKey& other) const {
+        return controller_id == other.controller_id &&
+               handle == other.handle &&
+               generation == other.generation;
+    }
+};
+
+struct HciConnectionKeyHash {
+    std::size_t operator()(const HciConnectionKey& k) const {
+        std::size_t h1 = std::hash<std::string>{}(k.controller_id);
+        std::size_t h2 = std::hash<uint16_t>{}(k.handle);
+        std::size_t h3 = std::hash<uint32_t>{}(k.generation);
+        return h1 ^ (h2 << 1) ^ (h3 << 2);
+    }
+};
+
+struct HciConnectionRecord {
+    HciConnectionKey key;
+    std::string device_mac;     // 大写规整 MAC，若未知为空
+    std::string address_type;   // "PUBLIC" | "RANDOM" | "UNKNOWN"
+    std::string attribution_scope; // "device" | "controller" | "unknown"
+    std::chrono::steady_clock::time_point connected_at;
+    std::optional<std::chrono::steady_clock::time_point> disconnected_at;
+    bool is_active{true};
+};
+

+    /**

+     * @param controller 控制器名（如 "hci0"）
+     * @param handle 12 位连接句柄（0..4095）
+     * @param mac 关联设备 MAC（可选，未知时传空）

+/*
+ * 文件: server/src/bpf/bt_hci_traffic.bpf.c
+ * 职责: 工业蓝牙 HCI ACL 数据通道观测（Phase 1 事实源）
+ *
+ * 核心设计:
+ *   1. 双向对称采集:
+ *      - TX: kprobe/hci_send_frame
+ *      - RX: kprobe/hci_recv_frame
+ *   2. 严格防御过滤:
+ *      - 在解引用数据前，读取 skb->cb 的 pkt_type，仅放行 HCI_ACLDATA_PKT (0x02)
+ *      - 严格校验 skb->len >= 4
+ *   3. 精准事实提取:
+ *      - handle: 提取低 12 位 (raw_handle & 0x0FFF)
+ *      - bytes:  累加 dlen (ACL 净荷字节数，不计物理层/HCI 传输开销)
+ *   4. 零音频语义、零硬编码结构体偏移，原生依赖内核网络核心 sk_buff (CO-RE 友好)
+ */
+
+#define __TARGET_ARCH_arm64
+#include "vmlinux.h"
+#include <bpf/bpf_helpers.h>
+#include <bpf/bpf_core_read.h>
+#include <bpf/bpf_tracing.h>
+
+char LICENSE[] SEC("license") = "GPL";
+
+#define HCI_ACLDATA_PKT 0x02
+
+struct bpf_acl_traffic_val {
+    __u64 tx_packets;
+    __u64 tx_bytes;
+    __u64 rx_packets;
+    __u64 rx_bytes;
+    __u64 last_tx_ktime_ns;
+    __u64 last_rx_ktime_ns;
+    __u64 non_acl_dropped;
+    __u64 short_pkt_dropped;
+};
+
+// 以 connection_handle (低 12 位，取值 0..4095) 为 Key 记录流量统计
+struct {
+    __uint(type, BPF_MAP_TYPE_HASH);
+    __uint(max_entries, 64);
+    __type(key, __u32);
+    __type(value, struct bpf_acl_traffic_val);
+} bt_acl_traffic SEC(".maps");
+
+// 全局开关与未归类统计
+struct {
+    __uint(type, BPF_MAP_TYPE_ARRAY);
+    __uint(max_entries, 4);
+    __type(key, __u32);
+    __type(value, __u64);
+} bt_acl_global_stats SEC(".maps");
+
+static __always_inline void bump_global_counter(__u32 idx, __u64 val) {
+    __u64 *p = bpf_map_lookup_elem(&bt_acl_global_stats, &idx);
+    if (p) {
+        __sync_fetch_and_add(p, val);
+    }
+}
+
+// ----------------------------------------------------------------------------
+// TX 挂点: kprobe/hci_send_frame(struct hci_dev *hdev, struct sk_buff *skb)
+// ----------------------------------------------------------------------------
+SEC("kprobe/hci_send_frame")
+int BPF_KPROBE(bt_hci_send_frame, void *hdev, struct sk_buff *skb) {
+    if (!skb) return 0;
+
+    // 1. 报文类型门控 (bt_cb(skb)->pkt_type 位于 skb->cb 第 0 字节)
+    __u8 pkt_type = 0;
+    if (bpf_probe_read_kernel(&pkt_type, sizeof(pkt_type),
+                              (const char *)skb + offsetof(struct sk_buff, cb)) < 0) {
+        return 0;
+    }
+
+    if (pkt_type != HCI_ACLDATA_PKT) {
+        bump_global_counter(0, 1); // TX 非 ACL 过滤
+        return 0;
+    }
+
+    // 2. 报文长度防御
+    __u32 len = BPF_CORE_READ(skb, len);
+    if (len < 4) {
+        bump_global_counter(2, 1); // TX 短报文丢弃
+        return 0;
+    }
+
+    const unsigned char *data = BPF_CORE_READ(skb, data);
+    if (!data) return 0;
+
+    // 3. 读取并解析 ACL 报头 (handle: 12bit, dlen: 16bit)
+    __u16 raw_handle = 0;
+    __u16 dlen = 0;
+    if (bpf_probe_read_kernel(&raw_handle, sizeof(raw_handle), data) < 0) return 0;
+    if (bpf_probe_read_kernel(&dlen, sizeof(dlen), data + 2) < 0) return 0;
+
+    __u32 handle = raw_handle & 0x0FFF;
+
+    // 4. 更新 Handle 统计
+    struct bpf_acl_traffic_val *stat = bpf_map_lookup_elem(&bt_acl_traffic, &handle);
+    if (!stat) {
+        struct bpf_acl_traffic_val init = {};
+        bpf_map_update_elem(&bt_acl_traffic, &handle, &init, BPF_NOEXIST);
+        stat = bpf_map_lookup_elem(&bt_acl_traffic, &handle);
+        if (!stat) return 0;
+    }
+
+    __u64 now = bpf_ktime_get_ns();
+    __sync_fetch_and_add(&stat->tx_packets, 1);
+    __sync_fetch_and_add(&stat->tx_bytes, dlen);
+    stat->last_tx_ktime_ns = now;
+
+    return 0;
+}
+
+// ----------------------------------------------------------------------------
+// RX 挂点: kprobe/hci_recv_frame(struct hci_dev *hdev, struct sk_buff *skb)
+// ----------------------------------------------------------------------------
+SEC("kprobe/hci_recv_frame")
+int BPF_KPROBE(bt_hci_recv_frame, void *hdev, struct sk_buff *skb) {
+    if (!skb) return 0;
+
+    // 1. 报文类型门控
+    __u8 pkt_type = 0;
+    if (bpf_probe_read_kernel(&pkt_type, sizeof(pkt_type),
+                              (const char *)skb + offsetof(struct sk_buff, cb)) < 0) {
+        return 0;
+    }
+
+    if (pkt_type != HCI_ACLDATA_PKT) {
+        bump_global_counter(1, 1); // RX 非 ACL 过滤 (如底噪 HCI Event)
+        return 0;
+    }
+
+    // 2. 报文长度防御
+    __u32 len = BPF_CORE_READ(skb, len);
+    if (len < 4) {
+        bump_global_counter(3, 1); // RX 短报文丢弃
+        return 0;
+    }
+
+    const unsigned char *data = BPF_CORE_READ(skb, data);
+    if (!data) return 0;
+
+    // 3. 读取并解析 ACL 报头
+    __u16 raw_handle = 0;
+    __u16 dlen = 0;
+    if (bpf_probe_read_kernel(&raw_handle, sizeof(raw_handle), data) < 0) return 0;
+    if (bpf_probe_read_kernel(&dlen, sizeof(dlen), data + 2) < 0) return 0;
+
+    __u32 handle = raw_handle & 0x0FFF;
+
+    // 4. 更新 Handle 统计
+    struct bpf_acl_traffic_val *stat = bpf_map_lookup_elem(&bt_acl_traffic, &handle);
+    if (!stat) {
+        struct bpf_acl_traffic_val init = {};
+        bpf_map_update_elem(&bt_acl_traffic, &handle, &init, BPF_NOEXIST);
+        stat = bpf_map_lookup_elem(&bt_acl_traffic, &handle);
+        if (!stat) return 0;
+    }
+
+    __u64 now = bpf_ktime_get_ns();
+    __sync_fetch_and_add(&stat->rx_packets, 1);
+    __sync_fetch_and_add(&stat->rx_bytes, dlen);
+    stat->last_rx_ktime_ns = now;
+
+    return 0;
+}

+bool BtDataPathAnalyzer::init(const std::string& bpfObjectPath, const std::string& controller) {
+    std::lock_guard<std::mutex> lock(mutex_);
+    controller_id_ = controller.empty() ? "hci0" : controller;
+    bpfObjectPath_ = bpfObjectPath;
+    lastError_.clear();
+    state_ = EbpfMonitorState::Initializing;
+    stateSupport_.setState(EbpfMonitorState::Initializing, false, "loading bt_hci_traffic BPF object");
+
+    bpfObj_ = bpf_object__open(bpfObjectPath.c_str());
+    if (!bpfObj_) {
+        lastError_ = "bpf_object__open failed for " + bpfObjectPath + ": " + std::string(strerror(errno));
+        LOG_WARNING(LogModule::BLUETOOTH, "BtDataPathAnalyzer: " << lastError_ << " — fallback mode");
+        state_ = EbpfMonitorState::Fallback;
+        stateSupport_.setState(EbpfMonitorState::Fallback, false, lastError_);
+        return false;
+    }
+
+    if (bpf_object__load(bpfObj_) != 0) {
+        lastError_ = "bpf_object__load failed: " + std::string(strerror(errno));
+        LOG_WARNING(LogModule::BLUETOOTH, "BtDataPathAnalyzer: " << lastError_ << " — fallback mode");
+        bpf_object__close(bpfObj_);
+        bpfObj_ = nullptr;
+        state_ = EbpfMonitorState::Fallback;
+        stateSupport_.setState(EbpfMonitorState::Fallback, false, lastError_);
+        return false;
+    }
+
+    mapFd_ = bpf_object__find_map_fd_by_name(bpfObj_, "bt_acl_traffic");
+    globalMapFd_ = bpf_object__find_map_fd_by_name(bpfObj_, "bt_acl_global_stats");
+
+    if (mapFd_ < 0) {
+        lastError_ = "Failed to find 'bt_acl_traffic' map in BPF object";
+        LOG_WARNING(LogModule::BLUETOOTH, "BtDataPathAnalyzer: " << lastError_);
+        bpf_object__close(bpfObj_);
+        bpfObj_ = nullptr;
+        state_ = EbpfMonitorState::Fallback;
+        stateSupport_.setState(EbpfMonitorState::Fallback, false, lastError_);
+        return false;
+    }
+
+    // 挂载 TX 与 RX kprobe 探针
+    bpf_program* prog_tx = bpf_object__find_program_by_name(bpfObj_, "bt_hci_send_frame");
+    bpf_program* prog_rx = bpf_object__find_program_by_name(bpfObj_, "bt_hci_recv_frame");
+
+    int attached_count = 0;
+    std::string hooks;
+
+    if (prog_tx) {
+        link_tx_ = bpf_program__attach(prog_tx);
+        if (!libbpf_get_error(link_tx_)) {
+            attached_count++;
+            hooks += "kprobe/hci_send_frame";
+        } else {
+            link_tx_ = nullptr;
+        }
+    }
+
+    if (prog_rx) {
+        link_rx_ = bpf_program__attach(prog_rx);
+        if (!libbpf_get_error(link_rx_)) {
+            attached_count++;
+            if (!hooks.empty()) hooks += ",";
+            hooks += "kprobe/hci_recv_frame";
+        } else {
+            link_rx_ = nullptr;
+        }
+    }
+
+    if (attached_count == 0) {
+        lastError_ = "All kprobe attach attempts failed (hci_send_frame, hci_recv_frame)";
+        LOG_WARNING(LogModule::BLUETOOTH, "BtDataPathAnalyzer: " << lastError_ << " — fallback mode");
+        bpf_object__close(bpfObj_);
+        bpfObj_ = nullptr;
+        mapFd_ = -1;
+        globalMapFd_ = -1;
+        state_ = EbpfMonitorState::Fallback;
+        stateSupport_.setState(EbpfMonitorState::Fallback, false, lastError_);
+        return false;
+    }
+
+    attachedHookName_ = hooks;
+    state_ = EbpfMonitorState::Attached;
+    stateSupport_.setState(EbpfMonitorState::Attached, true, attachedHookName_);
+    LOG_INFO(LogModule::BLUETOOTH, "BtDataPathAnalyzer attached successfully: " << attachedHookName_);
+    return true;
+}

+bool BtDataPathAnalyzer::readRawMapValue(uint16_t handle, RawAclTrafficVal* out) const {
+    if (!out) return false;
+    uint32_t key = handle & 0x0FFF;

+    if (mapFd_ >= 0) {

+        return (ret == 0);
+    }
+
+    // 内存测试桩回退
+    auto it = mock_map_.find(handle & 0x0FFF);
+    if (it != mock_map_.end()) {
+        *out = it->second;
+        return true;
+    }
+    return false;
+}

+std::vector<BtDataPathObservation> BtDataPathAnalyzer::collectObservations(
+    std::chrono::steady_clock::time_point now_point) {
+
+    std::vector<BtDataPathObservation> observations;
+    std::lock_guard<std::mutex> lock(mutex_);
+
+    // 1. 搜集当前存在的所有 Handle
+    std::vector<uint32_t> handles;
+
+    if (mapFd_ >= 0) {
+        uint32_t key = 0;
+        uint32_t next_key = 0;
+        int ret = bpf_map_get_next_key(mapFd_, nullptr, &next_key);
+        while (ret == 0) {
+            key = next_key;
+            handles.push_back(key);
+            ret = bpf_map_get_next_key(mapFd_, &key, &next_key);
+        }
+    } else {
+        for (const auto& [h, _] : mock_map_) {
+            handles.push_back(h);
+        }
+    }
+
+    // 同时补充 tracker 中当前活跃但内核 map 尚未建立条目的连接（保证 0 流量也观测到）
+    if (tracker_) {
+        auto active_conns = tracker_->getActiveConnections(controller_id_);
+        for (const auto& conn : active_conns) {
+            uint32_t h = conn.key.handle & 0x0FFF;
+            if (std::find(handles.begin(), handles.end(), h) == handles.end()) {
+                handles.push_back(h);
+            }
+        }
+    }
+
+    // 2. 逐 Handle 提取、核算滑动增量与速率
+    for (uint32_t raw_h : handles) {
+        uint16_t h = static_cast<uint16_t>(raw_h & 0x0FFF);
+        RawAclTrafficVal curr_val{};
+        readRawMapValue(h, &curr_val);
+
+        // 查找 Tracker 身份与代次
+        uint32_t current_generation = 0;
+        std::string attribution_scope = "unknown";
+        std::optional<std::string> device_id = std::nullopt;
+
+        if (tracker_) {
+            auto conn_rec = tracker_->getActiveConnection(controller_id_, h);
+            if (conn_rec.has_value()) {
+                current_generation = conn_rec->key.generation;
+                attribution_scope = conn_rec->attribution_scope;
+                if (!conn_rec->device_mac.empty()) {
+                    device_id = conn_rec->device_mac;
+                }
+            } else {
+                // 未知具体连接归属，但属于该 controller
+                attribution_scope = "controller";
+            }
+        }
+
+        // 增量计算
+        uint64_t tx_bytes = 0;
+        uint64_t rx_bytes = 0;
+        uint64_t tx_pkts = 0;
+        uint64_t rx_pkts = 0;
+        uint32_t window_ms = 0;
+        uint32_t max_gap_ms = 0;
+
+        auto last_it = last_samples_.find(h);
+        if (last_it != last_samples_.end()) {
+            // 防复用检查：若代次变化或累计计数回绕（如重启），重置增量基准
+            if (last_it->second.generation == current_generation &&
+                curr_val.tx_bytes >= last_it->second.val.tx_bytes &&
+                curr_val.rx_bytes >= last_it->second.val.rx_bytes) {
+
+                tx_bytes = curr_val.tx_bytes - last_it->second.val.tx_bytes;
+                rx_bytes = curr_val.rx_bytes - last_it->second.val.rx_bytes;
+                tx_pkts = curr_val.tx_packets - last_it->second.val.tx_packets;
+                rx_pkts = curr_val.rx_packets - last_it->second.val.rx_packets;
+
+                auto elapsed = std::chrono::duration_cast<std::chrono::milliseconds>(
+                    now_point - last_it->second.time).count();
+                window_ms = (elapsed > 0) ? static_cast<uint32_t>(elapsed) : 1;
+            } else {
+                // 代次跃迁，首次观测当前代次
+                tx_bytes = curr_val.tx_bytes;
+                rx_bytes = curr_val.rx_bytes;
+                tx_pkts = curr_val.tx_packets;
+                rx_pkts = curr_val.rx_packets;
+                window_ms = 1000;
+            }
+        } else {
+            // 首次观测
+            tx_bytes = curr_val.tx_bytes;
+            rx_bytes = curr_val.rx_bytes;
+            tx_pkts = curr_val.tx_packets;
+            rx_pkts = curr_val.rx_packets;
+            window_ms = 1000;
+        }
+
+        // 吞吐率计算（bits/s，ACL payload）
+        double tx_bps = 0.0;
+        double rx_bps = 0.0;
+        if (window_ms > 0) {
+            tx_bps = (static_cast<double>(tx_bytes) * 8.0 * 1000.0) / window_ms;
+            rx_bps = (static_cast<double>(rx_bytes) * 8.0 * 1000.0) / window_ms;
+        }
+
+        // 计算最后一次收发时间差（gap）
+        if (curr_val.last_tx_ktime_ns > 0 && curr_val.last_rx_ktime_ns > 0) {
+            uint64_t diff_ns = (curr_val.last_tx_ktime_ns > curr_val.last_rx_ktime_ns)
+                ? (curr_val.last_tx_ktime_ns - curr_val.last_rx_ktime_ns)
+                : (curr_val.last_rx_ktime_ns - curr_val.last_tx_ktime_ns);
+            max_gap_ms = static_cast<uint32_t>(diff_ns / 1000000ULL);
+        }
+
+        // 构造纯净 L1 DTO
+        BtDataPathObservation obs;
+        obs.observation_type = "BT_HCI_ACL_TRAFFIC";
+        obs.controller_id = controller_id_;
+        obs.connection_handle = h;
+        obs.connection_generation = current_generation;
+        obs.attribution_scope = attribution_scope;
+        obs.device_id = device_id;
+        obs.window_ms = window_ms;
+        obs.tx_bytes = tx_bytes;
+        obs.rx_bytes = rx_bytes;
+        obs.tx_packets = tx_pkts;
+        obs.rx_packets = rx_pkts;
+        obs.tx_throughput_bps = tx_bps;
+        obs.rx_throughput_bps = rx_bps;
+        obs.max_gap_ms = max_gap_ms;
+        obs.status = "OBSERVED";
+        obs.last_tx_ktime_ns = curr_val.last_tx_ktime_ns;
+        obs.last_rx_ktime_ns = curr_val.last_rx_ktime_ns;
+
+        observations.push_back(std::move(obs));
+
+        // 更新历史快照
+        last_samples_[h] = LastSample{curr_val, now_point, current_generation};
+    }
+
+    stateSupport_.recordReadSuccess(0, !observations.empty());
+    return observations;
+}

+HciConnectionKey BtHciConnectionTracker::onConnectionEstablished(
+    const std::string& controller,
+    uint16_t handle,
+    const std::string& mac,
+    const std::string& addr_type) {
+
+    uint16_t valid_handle = handle & 0x0FFF;
+    std::string slot_key = makeSlotKey(controller, valid_handle);
+    std::string clean_mac = normalizeMac(mac);
+
+    std::lock_guard<std::mutex> lock(mutex_);
+
+    auto it = active_map_.find(slot_key);
+    if (it != active_map_.end() && it->second.is_active) {
+        // 重复通知幂等处理：若 MAC 一致，直接复用当前代次
+        if (!clean_mac.empty() && it->second.device_mac == clean_mac) {
+            return it->second.key;
+        }
+        // 若 MAC 发生冲突或前连接未正常收到 disconnect，强制递增代次
+        it->second.is_active = false;
+        it->second.disconnected_at = std::chrono::steady_clock::now();
+    }
+
+    uint32_t next_gen = ++generation_counter_[slot_key];
+
+    HciConnectionKey key{controller, valid_handle, next_gen};
+    HciConnectionRecord record;
+    record.key = key;
+    record.device_mac = clean_mac;
+    record.address_type = addr_type.empty() ? "UNKNOWN" : addr_type;
+    record.attribution_scope = clean_mac.empty() ? "controller" : "device";
+    record.connected_at = std::chrono::steady_clock::now();
+    record.is_active = true;
+
+    active_map_[slot_key] = record;
+    return key;
+}

+bool BtHciConnectionTracker::onConnectionDisconnected(
+    const std::string& controller, uint16_t handle) {
+
+    uint16_t valid_handle = handle & 0x0FFF;
+    std::string slot_key = makeSlotKey(controller, valid_handle);
+
+    std::lock_guard<std::mutex> lock(mutex_);
+    auto it = active_map_.find(slot_key);
+    if (it == active_map_.end() || !it->second.is_active) {
+        return false;
+    }
+
+    it->second.is_active = false;
+    it->second.disconnected_at = std::chrono::steady_clock::now();
+    return true;

+#define __TARGET_ARCH_arm64
+#include "vmlinux.h"
+#include <bpf/bpf_helpers.h>
+#include <bpf/bpf_core_read.h>
+#include <bpf/bpf_tracing.h>
+
+char LICENSE[] SEC("license") = "GPL";
+
+#define HCI_ACLDATA_PKT 0x02
+
+struct acl_stat {
+    __u64 tx_pkts;
+    __u64 tx_bytes;
+    __u64 rx_pkts;
+    __u64 rx_bytes;
+    __u64 non_acl_dropped;
+    __u64 short_pkt_dropped;
+};
+
+// 以 handle (低 12 位) 为 key 聚合统计
+struct {
+    __uint(type, BPF_MAP_TYPE_HASH);
+    __uint(max_entries, 64);
+    __type(key, __u32);
+    __type(value, struct acl_stat);
+} hci_acl_stats SEC(".maps");
+
+// 全局底噪与非 ACL 计数（Key 0: TX non-acl, Key 1: RX non-acl）
+struct {
+    __uint(type, BPF_MAP_TYPE_ARRAY);
+    __uint(max_entries, 4);
+    __type(key, __u32);
+    __type(value, __u64);
+} global_drop_stats SEC(".maps");
+
+// 辅助递增全局 drop
+static __always_inline void bump_global_drop(__u32 idx) {
+    __u64 *val = bpf_map_lookup_elem(&global_drop_stats, &idx);
+    if (val) {
+        __sync_fetch_and_add(val, 1);
+    }
+}
+
+// ----------------------------------------------------------------------------
+// TX 挂点: kprobe/hci_send_frame(struct hci_dev *hdev, struct sk_buff *skb)
+// ----------------------------------------------------------------------------
+SEC("kprobe/hci_send_frame")
+int BPF_KPROBE(probe_hci_send_frame, void *hdev, struct sk_buff *skb) {
+    if (!skb) return 0;
+
+    // 1. 检查报文类型: 蓝牙子系统将 pkt_type 存放在 skb->cb 中 (bt_cb(skb)->pkt_type)
+    // 根据内核定义，bt_skb_cb.pkt_type 位于 skb->cb 的第 0 字节
+    __u8 pkt_type = 0;
+    if (bpf_probe_read_kernel(&pkt_type, sizeof(pkt_type), (const char *)skb + offsetof(struct sk_buff, cb)) < 0) {
+        return 0;
+    }
+
+    if (pkt_type != HCI_ACLDATA_PKT) {
+        bump_global_drop(0); // TX 非 ACL 丢弃计数
+        return 0;
+    }
+
+    // 2. 检查 skb 长度与数据指针
+    __u32 len = BPF_CORE_READ(skb, len);
+    if (len < 4) {
+        bump_global_drop(2); // TX 过短报文
+        return 0;
+    }
+
+    const unsigned char *data = BPF_CORE_READ(skb, data);
+    if (!data) return 0;
+
+    // 3. 读取 HCI ACL Header
+    __u16 raw_handle = 0;
+    __u16 dlen = 0;
+    if (bpf_probe_read_kernel(&raw_handle, sizeof(raw_handle), data) < 0) return 0;
+    if (bpf_probe_read_kernel(&dlen, sizeof(dlen), data + 2) < 0) return 0;
+
+    __u32 handle = raw_handle & 0x0FFF;
+
+    // 4. 统计更新
+    struct acl_stat *stat = bpf_map_lookup_elem(&hci_acl_stats, &handle);
+    if (!stat) {
+        struct acl_stat zero = {};
+        bpf_map_update_elem(&hci_acl_stats, &handle, &zero, BPF_NOEXIST);
+        stat = bpf_map_lookup_elem(&hci_acl_stats, &handle);
+        if (!stat) return 0;
+    }
+
+    __sync_fetch_and_add(&stat->tx_pkts, 1);
+    __sync_fetch_and_add(&stat->tx_bytes, dlen);
+    return 0;
+}
+
+// ----------------------------------------------------------------------------
+// RX 挂点: kprobe/hci_recv_frame(struct hci_dev *hdev, struct sk_buff *skb)
+// ----------------------------------------------------------------------------
+SEC("kprobe/hci_recv_frame")
+int BPF_KPROBE(probe_hci_recv_frame, void *hdev, struct sk_buff *skb) {
+    if (!skb) return 0;
+
+    // 1. 检查报文类型
+    __u8 pkt_type = 0;
+    if (bpf_probe_read_kernel(&pkt_type, sizeof(pkt_type), (const char *)skb + offsetof(struct sk_buff, cb)) < 0) {
+        return 0;
+    }
+
+    if (pkt_type != HCI_ACLDATA_PKT) {
+        bump_global_drop(1); // RX 非 ACL 丢弃计数 (例如海量的 HCI Event)
+        return 0;
+    }
+
+    // 2. 检查 skb 长度
+    __u32 len = BPF_CORE_READ(skb, len);
+    if (len < 4) {
+        bump_global_drop(3); // RX 过短报文
+        return 0;
+    }
+
+    const unsigned char *data = BPF_CORE_READ(skb, data);
+    if (!data) return 0;
+
+    // 3. 读取 HCI ACL Header
+    __u16 raw_handle = 0;
+    __u16 dlen = 0;
+    if (bpf_probe_read_kernel(&raw_handle, sizeof(raw_handle), data) < 0) return 0;
+    if (bpf_probe_read_kernel(&dlen, sizeof(dlen), data + 2) < 0) return 0;
+
+    __u32 handle = raw_handle & 0x0FFF;
+
+    // 4. 统计更新
+    struct acl_stat *stat = bpf_map_lookup_elem(&hci_acl_stats, &handle);
+    if (!stat) {
+        struct acl_stat zero = {};
+        bpf_map_update_elem(&hci_acl_stats, &handle, &zero, BPF_NOEXIST);
+        stat = bpf_map_lookup_elem(&hci_acl_stats, &handle);
+        if (!stat) return 0;
+    }
+
+    __sync_fetch_and_add(&stat->rx_pkts, 1);
+    __sync_fetch_and_add(&stat->rx_bytes, dlen);
+    return 0;
+}
