+size_t BtHciConnectionTracker::syncFromKernel(const std::string& controller) {
+    // 1. 打开 HCI 控制套接字 (AF_BLUETOOTH, SOCK_RAW, BTPROTO_HCI)
+    #ifndef AF_BLUETOOTH
+    #define AF_BLUETOOTH 31
+    #endif
+    #ifndef BTPROTO_HCI
+    #define BTPROTO_HCI 1
+    #endif
+    #ifndef HCIGETCONNLIST
+    // _IOR('H', 212, int) = 0x800448d4
+    #define HCIGETCONNLIST 0x800448d4
+    #endif
+
+    int sock = ::socket(AF_BLUETOOTH, SOCK_RAW | SOCK_CLOEXEC, BTPROTO_HCI);
+    if (sock < 0) {
+        return activeCount();
+    }
+
+    // 解析 controller 编号，如 "hci0" -> 0
+    uint16_t dev_id = 0;
+    if (controller.rfind("hci", 0) == 0 && controller.size() > 3) {
+        try {
+            dev_id = static_cast<uint16_t>(std::stoi(controller.substr(3)));
+        } catch (...) {
+            dev_id = 0;
+        }
+    }
+
+    // 内存缓冲区：头部 4 字节 (dev_id + conn_num) + 最多 64 条 struct hci_conn_info (每条 16 字节)
+    struct RawHciConnInfo {
+        uint16_t handle;
+        uint8_t bdaddr[6];
+        uint8_t type;
+        uint8_t out;
+        uint16_t state;
+        uint32_t link_mode;
+    } __attribute__((packed));
+
+    uint8_t buf[4 + 64 * sizeof(RawHciConnInfo)] = {0};
+    *reinterpret_cast<uint16_t*>(buf) = dev_id;
+    *reinterpret_cast<uint16_t*>(buf + 2) = 64; // 最多获取 64 个连接
+
+    int ret = ::ioctl(sock, HCIGETCONNLIST, buf);
+    ::close(sock);
+
+    if (ret < 0) {
+        return activeCount();
+    }
+
+    uint16_t conn_num = *reinterpret_cast<uint16_t*>(buf + 2);
+    if (conn_num > 64) conn_num = 64;
+
+    std::set<uint16_t> live_handles;
+    const RawHciConnInfo* conns = reinterpret_cast<const RawHciConnInfo*>(buf + 4);
+
+    for (uint16_t i = 0; i < conn_num; ++i) {
+        uint16_t handle = conns[i].handle & 0x0FFF;
+        live_handles.insert(handle);
+
+        // 内核 bdaddr 反序存储 (b[0] 为低位)

+                 conns[i].bdaddr[2], conns[i].bdaddr[1], conns[i].bdaddr[0]);

+bool DbusService::handleQueryBtDataPath(DBusConnection* conn, DBusMessage* msg) {
+    LOG_INFO(LogModule::DBUS, "handleQueryBtDataPath called");
+
+    // 参数：STRING device_address, INT64 start_ms, INT64 end_ms, INT32 limit
+    std::string device_address;
+    int64_t start_ms = 0, end_ms = 0;
+    int32_t limit = 100;
+
+    DBusMessageIter args;
+    if (dbus_message_iter_init(msg, &args) == TRUE) {
+        if (dbus_message_iter_get_arg_type(&args) == DBUS_TYPE_STRING) {
+            const char* val = nullptr;
+            dbus_message_iter_get_basic(&args, &val);
+            if (val) device_address = val;
+        }
+        if (dbus_message_iter_next(&args) &&
+            dbus_message_iter_get_arg_type(&args) == DBUS_TYPE_INT64) {
+            dbus_message_iter_get_basic(&args, &start_ms);
+        }
+        if (dbus_message_iter_next(&args) &&
+            dbus_message_iter_get_arg_type(&args) == DBUS_TYPE_INT64) {
+            dbus_message_iter_get_basic(&args, &end_ms);
+        }
+        if (dbus_message_iter_next(&args) &&
+            dbus_message_iter_get_arg_type(&args) == DBUS_TYPE_INT32) {
+            dbus_message_iter_get_basic(&args, &limit);
+        }
+    }
+
+    constexpr int32_t kMaxLimit = 10000;
+    if (limit <= 0) {
+        limit = 1;
+    } else if (limit > kMaxLimit) {
+        limit = kMaxLimit;
+    }
+    if (start_ms > 0 && end_ms > 0 && end_ms < start_ms) {
+        std::swap(start_ms, end_ms);
+    }
+
+    std::string result = "[]";
+    if (ctx_ && ctx_->db_mgr) {
+        result = ctx_->db_mgr->queryBtDataPathMetrics(device_address, start_ms, end_ms, limit);
+    } else {
+        result = "{\"error\":\"database not available\"}";
+    }
+
+    DBusMessage* reply = dbus_message_new_method_return(msg);
+    if (!reply) return false;
+
+    DBusMessageIter out_args;
+    dbus_message_iter_init_append(reply, &out_args);
+    const char* str = result.c_str();
+    if (!dbus_message_iter_append_basic(&out_args, DBUS_TYPE_STRING, &str)) {
+        dbus_message_unref(reply);
+        return false;
+    }
+
+    dbus_connection_send(conn, reply, nullptr);
+    dbus_connection_flush(conn);
+    dbus_message_unref(reply);
+    return true;
+}
