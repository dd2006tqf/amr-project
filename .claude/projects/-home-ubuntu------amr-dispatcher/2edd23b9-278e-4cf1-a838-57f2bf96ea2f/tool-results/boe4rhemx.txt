    71	std::optional<double> RouteOrEuclideanDistance(
    72	    const StationCatalog& catalog, const std::string& from, const std::string& to) {
    73	  if (from.empty() || to.empty()) {
    74	    return std::nullopt;
    75	  }
    76	  if (const auto route_distance = EstimateRouteDistance(catalog, from, to)) {
    77	    return route_distance;
    78	  }
    79	  const auto* from_station = FindStation(catalog, from);
    80	  const auto* to_station = FindStation(catalog, to);
    81	  if (from_station == nullptr || to_station == nullptr ||
    82	      from_station->frame_id != to_station->frame_id) {
    83	    return std::nullopt;
    84	  }
    85	  return std::hypot(from_station->x - to_station->x, from_station->y - to_station->y);
    86	}

    90	std::optional<FleetCatalog> LoadFleetCatalog(const std::filesystem::path& path) {
    91	  std::ifstream input(path);
    92	  if (!input) {
    93	    return std::nullopt;
    94	  }
    95	
    96	  FleetCatalog catalog;
    97	  FleetRobot current;
    98	  bool in_robot = false;
    99	  bool has_id = false;
   100	  std::string section;
   101	  std::string line;
   102	
   103	  auto flush_robot = [&]() {
   104	    if (in_robot && has_id) {
   105	      catalog.robots.push_back(current);
   106	    }
   107	    current = FleetRobot{};
   108	    in_robot = false;
   109	    has_id = false;
   110	  };
   111	
   112	  while (std::getline(input, line)) {
   113	    line = Trim(line);
   114	    if (line.empty() || line.front() == '#') {
   115	      continue;
   116	    }
   117	    if (line == "robots:" || line == "fleet:") {
   118	      flush_robot();
   119	      section = line.substr(0, line.size() - 1);
   120	      continue;
   121	    }
   122	    if (line.rfind("- ", 0) == 0) {
   123	      if (section != "robots" && section != "fleet") {
   124	        continue;
   125	      }
   126	      flush_robot();
   127	      in_robot = true;
   128	      line = Trim(line.substr(2));
   129	    }
   130	    if (!in_robot) {
   131	      continue;
   132	    }
   133	    if (auto value = ValueAfterColon(line, "id")) {
   134	      current.id = *value;
   135	      has_id = !current.id.empty();
   136	      continue;
   137	    }
   138	    if (auto value = ValueAfterColon(line, "enabled")) {
   139	      current.enabled = ParseBool(*value);
   140	      continue;
   141	    }
   142	    if (auto value = ValueAfterColon(line, "state")) {
   143	      current.state = value->empty() ? "available" : *value;
   144	      continue;
   145	    }
   146	    if (auto value = ValueAfterColon(line, "battery_voltage")) {
   147	      if (auto parsed = ParseDouble(*value)) {
   148	        current.battery_voltage = *parsed;
   149	      }
   150	      continue;
   151	    }
   152	    if (auto value = ValueAfterColon(line, "max_payload_kg")) {
   153	      if (auto parsed = ParseDouble(*value)) {
   154	        current.max_payload_kg = *parsed;
   155	      }
   156	      continue;
   157	    }
   158	    if (auto value = ValueAfterColon(line, "current_station_id")) {
   159	      current.current_station_id = *value;
   160	      continue;
   161	    }
   162	    if (auto value = ValueAfterColon(line, "capabilities")) {
   163	      current.capabilities = SplitCommaList(*value);
   164	      continue;
   165	    }
   166	    if (auto value = ValueAfterColon(line, "backend")) {
   167	      // 执行后端（P0.1）：只接受已知标签，未知值回落 local_sim 并由校验报错
   168	      current.backend = value->empty() ? "local_sim" : *value;
   169	      continue;
   170	    }
   171	  }
   172	  flush_robot();
   173	
   174	  std::string message;
   175	  if (!ValidateFleetCatalog(catalog, &message)) {
   176	    return std::nullopt;
   177	  }
   178	  return catalog;
   179	}

   198	const FleetRobot* SelectFleetRobot(
   199	    const FleetCatalog& catalog, const std::string& required_capability,
   200	    const std::vector<std::string>& unavailable_robot_ids) {
   201	  const FleetRobot* selected = nullptr;
   202	  for (const auto& robot : catalog.robots) {
   203	    if (!RobotAvailableForTask(robot) || !RobotHasCapability(robot, required_capability)) {
   204	      continue;
   205	    }
   206	    if (RobotUnavailable(robot, unavailable_robot_ids)) {
   207	      continue;
   208	    }
   209	    if (selected == nullptr || robot.battery_voltage > selected->battery_voltage) {
   210	      selected = &robot;
   211	    }
   212	  }
   213	  return selected;
   214	}

   218	const FleetRobot* SelectFleetRobotForStationTask(
   219	    const FleetCatalog& fleet_catalog, const StationCatalog& station_catalog,
   220	    const std::string& required_capability, const std::string& pickup_station,
   221	    const std::string& dropoff_station, const std::vector<std::string>& unavailable_robot_ids,
   222	    FleetRobotTaskScore* selected_score) {
   223	  const auto task_distance = RouteOrEuclideanDistance(station_catalog, pickup_station, dropoff_station);
   224	  if (!task_distance.has_value()) {
   225	    return SelectFleetRobot(fleet_catalog, required_capability, unavailable_robot_ids);
   226	  }
   227	
   228	  const FleetRobot* selected = nullptr;
   229	  FleetRobotTaskScore best_score;
   230	  for (const auto& robot : fleet_catalog.robots) {
   231	    if (!RobotAvailableForTask(robot) || !RobotHasCapability(robot, required_capability) ||
   232	        RobotUnavailable(robot, unavailable_robot_ids)) {
   233	      continue;
   234	    }
   235	    const auto approach_distance =
   236	        RouteOrEuclideanDistance(station_catalog, robot.current_station_id, pickup_station);
   237	    if (!approach_distance.has_value()) {
   238	      continue;
   239	    }
   240	    FleetRobotTaskScore score;
   241	    score.robot_id = robot.id;
   242	    score.approach_distance_m = *approach_distance;
   243	    score.task_distance_m = *task_distance;
   244	    score.total_distance_m = score.approach_distance_m + score.task_distance_m;
   245	    score.battery_voltage = robot.battery_voltage;
   246	    if (selected == nullptr || score.total_distance_m < best_score.total_distance_m ||
   247	        (score.total_distance_m == best_score.total_distance_m &&
   248	         score.battery_voltage > best_score.battery_voltage)) {
   249	      selected = &robot;
   250	      best_score = score;
   251	    }
   252	  }
   253	  if (selected != nullptr && selected_score != nullptr) {
   254	    *selected_score = best_score;
   255	  }
   256	  if (selected == nullptr) {
   257	    return SelectFleetRobot(fleet_catalog, required_capability, unavailable_robot_ids);
   258	  }
   259	  return selected;
   260	}

   262	std::optional<double> EstimateVehicleApproachCost(
   263	    const StationCatalog& catalog, const double vehicle_x, const double vehicle_y,
   264	    const std::string& vehicle_station_id, const std::string& target_station_id) {
   265	  if (target_station_id.empty()) {
   266	    return std::nullopt;
   267	  }
   268	  const auto* target = FindStation(catalog, target_station_id);
   269	  if (target == nullptr) {
   270	    return std::nullopt;
   271	  }
   272	  // 规则 1：车挂在已知站点上 -> 直接算路网成本
   273	  if (!vehicle_station_id.empty()) {
   274	    return EstimateRouteDistance(catalog, vehicle_station_id, target_station_id);
   275	  }
   276	  // 规则 2：车在站点之间 -> 先直线接驳到最近站点，再叠路网成本
   277	  const Station* nearest = nullptr;
   278	  double nearest_dist = 0.0;
   279	  for (const auto& station : catalog.stations) {
   280	    if (station.frame_id != target->frame_id) {
   281	      continue;  // 跨 frame 不比较
   282	    }
   283	    const double d = std::hypot(station.x - vehicle_x, station.y - vehicle_y);
   284	    if (nearest == nullptr || d < nearest_dist) {
   285	      nearest = &station;
   286	      nearest_dist = d;
   287	    }
   288	  }
   289	  if (nearest == nullptr) {
   290	    return std::nullopt;
   291	  }
   292	  const auto route = EstimateRouteDistance(catalog, nearest->id, target_station_id);
   293	  if (!route.has_value()) {
   294	    return std::nullopt;
   295	  }
   296	  return nearest_dist + *route;
   297	}

   299	bool ValidateFleetCatalog(const FleetCatalog& catalog, std::string* message) {
   300	  std::unordered_set<std::string> ids;
   301	  for (const auto& robot : catalog.robots) {
   302	    if (robot.id.empty()) {
   303	      if (message != nullptr) {
   304	        *message = "fleet robot id is empty";
   305	      }
   306	      return false;
   307	    }
   308	    if (!ids.insert(robot.id).second) {
   309	      if (message != nullptr) {
   310	        *message = "duplicate fleet robot id: " + robot.id;
   311	      }
   312	      return false;
   313	    }
   314	    if (!std::isfinite(robot.battery_voltage) || robot.battery_voltage < 0.0) {
   315	      if (message != nullptr) {
   316	        *message = "invalid battery voltage for fleet robot: " + robot.id;
   317	      }
   318	      return false;
   319	    }
   320	    if (!std::isfinite(robot.max_payload_kg) || robot.max_payload_kg < 0.0) {
   321	      if (message != nullptr) {
   322	        *message = "invalid payload for fleet robot: " + robot.id;
   323	      }
   324	      return false;
   325	    }
   326	    // 执行后端标签必须是已知值（local_sim / vda5050）——不猜、不静默回落
   327	    if (robot.backend != "local_sim" && robot.backend != "vda5050") {
   328	      if (message != nullptr) {
   329	        *message = "unknown execution backend for fleet robot " + robot.id + ": " + robot.backend;
   330	      }
   331	      return false;
   332	    }
   333	  }
   334	  if (message != nullptr) {
   335	    *message = "fleet catalog ok";
   336	  }
   337	  return true;
   338	}

   206	  const std::size_t before = reload_count_;

   208	  if (!loaded_) {

   222	void FacilityCatalogReloader::TryLoad(bool count_as_reload) {
   223	  if (path_.empty()) {
   224	    return;
   225	  }
   226	  std::error_code ec;
   227	  if (!std::filesystem::exists(path_, ec)) {
   228	    last_error_ = "facility catalog file does not exist: " + path_.string();
   229	    return;
   230	  }
   231	
   232	  const auto loaded = LoadFacilityCatalog(path_);
   233	  if (!loaded.has_value()) {
   234	    last_error_ = "failed to load/validate facility catalog from: " + path_.string();
   235	    return;
   236	  }
   237	
   238	  catalog_ = *loaded;
   239	  loaded_ = true;
   240	  last_error_.clear();
   241	  if (count_as_reload) {
   242	    ++reload_count_;
   243	  }
   244	}

    15	std::string Trim(const std::string& value) {
    16	  const auto first = value.find_first_not_of(" \t\r\n");
    17	  if (first == std::string::npos) {
    18	    return "";
    19	  }
    20	  const auto last = value.find_last_not_of(" \t\r\n");
    21	  return value.substr(first, last - first + 1);
    22	}

    24	std::optional<std::string> ValueAfterColon(const std::string& line, const std::string& key) {
    25	  const std::string prefix = key + ":";
    26	  if (line.rfind(prefix, 0) != 0) {
    27	    return std::nullopt;
    28	  }
    29	  return Trim(line.substr(prefix.size()));
    30	}

    33	  return value == "true" || value == "True" || value == "1";

    38	std::optional<DockCatalog> LoadDockCatalog(const std::filesystem::path& path) {
    39	  std::ifstream input(path);
    40	  if (!input) {
    41	    return std::nullopt;
    42	  }
    43	
    44	  DockCatalog catalog;
    45	  Dock current;
    46	  bool in_dock = false;
    47	  bool has_id = false;
    48	  bool has_station = false;
    49	  bool has_charger = false;
    50	
    51	  auto flush_dock = [&]() {
    52	    if (in_dock && has_id && has_station && has_charger) {
    53	      if (current.approach_station_id.empty()) {
    54	        current.approach_station_id = current.station_id;
    55	      }
    56	      catalog.docks.push_back(current);
    57	    }
    58	    current = Dock{};
    59	    in_dock = false;
    60	    has_id = false;
    61	    has_station = false;
    62	    has_charger = false;
    63	  };
    64	
    65	  std::string section;
    66	  std::string line;
    67	  while (std::getline(input, line)) {
    68	    line = Trim(line);
    69	    if (line.empty() || line.front() == '#') {
    70	      continue;
    71	    }
    72	    if (line == "docks:") {
    73	      flush_dock();
    74	      section = "docks";
    75	      continue;
    76	    }
    77	    if (line.rfind("- ", 0) == 0) {
    78	      if (section != "docks") {
    79	        continue;
    80	      }
    81	      flush_dock();
    82	      in_dock = true;
    83	      line = Trim(line.substr(2));
    84	    }
    85	    if (!in_dock) {
    86	      continue;
    87	    }
    88	    if (auto value = ValueAfterColon(line, "id")) {
    89	      current.id = *value;
    90	      has_id = !current.id.empty();
    91	      continue;
    92	    }
    93	    if (auto value = ValueAfterColon(line, "station_id")) {
    94	      current.station_id = *value;
    95	      has_station = !current.station_id.empty();
    96	      continue;
    97	    }
    98	    if (auto value = ValueAfterColon(line, "approach_station_id")) {
    99	      current.approach_station_id = *value;
   100	      continue;
   101	    }
   102	    if (auto value = ValueAfterColon(line, "charger_resource_id")) {
   103	      current.charger_resource_id = *value;
   104	      has_charger = !current.charger_resource_id.empty();
   105	      continue;
   106	    }
   107	    if (auto value = ValueAfterColon(line, "enabled")) {
   108	      current.enabled = ParseBool(*value);
   109	      continue;
   110	    }
   111	  }
   112	  flush_dock();
   113	
   114	  std::string message;
   115	  if (!ValidateDockCatalog(catalog, &message)) {
   116	    return std::nullopt;
   117	  }
   118	  return catalog;
   119	}

   139	bool ValidateDockCatalog(const DockCatalog& catalog, std::string* message) {
   140	  std::unordered_set<std::string> ids;
   141	  for (const auto& dock : catalog.docks) {
   142	    if (dock.id.empty()) {
   143	      if (message != nullptr) {
   144	        *message = "dock id is empty";
   145	      }
   146	      return false;
   147	    }
   148	    if (dock.station_id.empty()) {
   149	      if (message != nullptr) {
   150	        *message = "dock station_id is empty: " + dock.id;
   151	      }
   152	      return false;
   153	    }
   154	    if (dock.charger_resource_id.empty()) {
   155	      if (message != nullptr) {
   156	        *message = "dock charger_resource_id is empty: " + dock.id;
   157	      }
   158	      return false;
   159	    }
   160	    if (!ids.insert(dock.id).second) {
   161	      if (message != nullptr) {
   162	        *message = "duplicate dock id: " + dock.id;
   163	      }
   164	      return false;
   165	    }
   166	  }
   167	  if (message != nullptr) {
   168	    *message = "dock catalog ok";
   169	  }
   170	  return true;
   171	}

   173	DockStateProjection BuildDockStateProjection(
   174	    const Dock& dock, const DockingStateMap& dock_states,
   175	    const FacilityReservationMap& reservations) {
   176	  DockStateProjection projection;
   177	  projection.dock_id = dock.id;
   178	  projection.station_id = dock.station_id;
   179	  projection.approach_station_id = dock.approach_station_id;
   180	  projection.charger_resource_id = dock.charger_resource_id;
   181	  projection.enabled = dock.enabled;
   182	  projection.available = dock.enabled && reservations.find(dock.charger_resource_id) == reservations.end();
   183	  projection.state = dock.enabled ? "AVAILABLE" : "DISABLED";
   184	
   185	  const auto reservation = reservations.find(dock.charger_resource_id);
   186	  if (reservation != reservations.end()) {
   187	    projection.state = reservation->second.status;
   188	    projection.mission_id = reservation->second.mission_id;
   189	  }
   190	  const auto runtime = dock_states.find(dock.id);
   191	  if (runtime != dock_states.end()) {
   192	    projection.state = runtime->second.state;
   193	    projection.mission_id = runtime->second.mission_id;
   194	  }
   195	  return projection;
   196	}

   198	DockListProjection BuildDockListProjection(
   199	    const DockCatalog& catalog, const DockingStateMap& dock_states,
   200	    const FacilityReservationMap& reservations, const bool include_disabled) {
   201	  DockListProjection projection;
   202	  projection.dock_ids.reserve(catalog.docks.size());
   203	  projection.station_ids.reserve(catalog.docks.size());
   204	  projection.approach_station_ids.reserve(catalog.docks.size());
   205	  projection.charger_resource_ids.reserve(catalog.docks.size());
   206	  projection.enabled.reserve(catalog.docks.size());
   207	  projection.available.reserve(catalog.docks.size());
   208	  projection.states.reserve(catalog.docks.size());
   209	  projection.mission_ids.reserve(catalog.docks.size());
   210	  for (const auto& dock : catalog.docks) {
   211	    if (!include_disabled && !dock.enabled) {
   212	      continue;
   213	    }
   214	    const auto state = BuildDockStateProjection(dock, dock_states, reservations);
   215	    projection.dock_ids.push_back(state.dock_id);
   216	    projection.station_ids.push_back(state.station_id);
   217	    projection.approach_station_ids.push_back(state.approach_station_id);
   218	    projection.charger_resource_ids.push_back(state.charger_resource_id);
   219	    projection.enabled.push_back(state.enabled);
   220	    projection.available.push_back(state.available);
   221	    projection.states.push_back(state.state);
   222	    projection.mission_ids.push_back(state.mission_id);
   223	  }
   224	  projection.message = "loaded " + std::to_string(projection.dock_ids.size()) + " dock(s)";
   225	  return projection;
   226	}
