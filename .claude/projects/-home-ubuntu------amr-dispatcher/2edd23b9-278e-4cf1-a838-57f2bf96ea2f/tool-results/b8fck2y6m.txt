     1	#!/usr/bin/env bash
     2	set -euo pipefail
     3	
     4	SCRIPT_DIR="$(cd "$(dirname "${BASH_SOURCE[0]}")" && pwd)"
     5	WORKSPACE_DIR="$(cd "${SCRIPT_DIR}/.." && pwd)"
     6	REPORT_FILE="${WORKSPACE_DIR}/test_report_matrix.json"
     7	
     8	START_TIME=$(date +%s)
     9	
    10	echo "================================================================="
    11	echo "       AMR Dispatcher 工业级系统基础设施与混沌演练流水线         "
    12	echo "================================================================="
    13	
    14	# ---------------------------------------------------------------------------
    15	# 阶段状态一律由实际执行结果生成：PASSED / FAILED / SKIPPED。
    16	#   PASSED  = 阶段执行成功
    17	#   FAILED  = 阶段执行失败（计入失败数，脚本最终以非零退出）
    18	#   SKIPPED = 本机不具备执行条件（不计入失败，但如实写入报告与 verdict）
    19	# 矩阵中不再写入任何硬编码的通过结论；verdict 只基于实际执行的阶段。
    20	# ---------------------------------------------------------------------------
    21	CORE_STATUS="SKIPPED"
    22	BENCH_STATUS="SKIPPED"
    23	BT_STATUS="SKIPPED"
    24	CHAOS_DEADLOCK_STATUS="SKIPPED"
    25	CHAOS_WATCHDOG_STATUS="SKIPPED"
    26	CHAOS_FAILOVER_STATUS="SKIPPED"
    27	# L2 契约 / 运维验收：需要容器内的 ROS 运行时（Docker + colcon 构建产物）。
    28	# 宿主机没有则如实记 SKIPPED，绝不写成 PASSED。
    29	L2_CONTRACTS_STATUS="SKIPPED"
    30	OPS_SUPERVISION_STATUS="SKIPPED"
    31	OPS_TOPOLOGY_STATUS="SKIPPED"
    32	FAILED_STAGES=0
    33	
    34	# record_status <目标变量名> <PASSED|FAILED>
    35	record_status() {
    36	  local -n _status_ref="$1"
    37	  _status_ref="$2"
    38	  if [ "$2" = "FAILED" ]; then
    39	    FAILED_STAGES=$((FAILED_STAGES + 1))
    40	  fi
    41	}
    42	
    43	# count_suites <目录...>：统计测试源码中唯一 TEST/TEST_F 套件名数量（无匹配时输出 0）
    44	count_suites() {
    45	  {
    46	    grep -rhoE 'TEST(_F)?[[:space:]]*\([[:space:]]*[A-Za-z0-9_]+' "$@" 2>/dev/null \
    47	      | sed -E 's/.*\([[:space:]]*//' \
    48	      | sort -u \
    49	      | wc -l \
    50	      | tr -d '[:space:]'
    51	  } || true
    52	}
    53	
    54	# count_ctest_cases <构建目录>：读取 ctest 注册的用例总数（仅列举不执行）
    55	count_ctest_cases() {
    56	  {
    57	    ctest --test-dir "$1" -N 2>/dev/null \
    58	      | sed -n 's/.*Total Tests: \([0-9][0-9]*\).*/\1/p' \
    59	      | tail -1
    60	  } || true
    61	}
    62	
    63	# json_number <值>：空值降级为 JSON null
    64	json_number() {
    65	  if [ -n "$1" ]; then printf '%s' "$1"; else printf 'null'; fi
    66	}
    67	
    68	# ---------------------------------------------------------------------------
    69	# L1 核心层：纯 C++ 编译 + GoogleTest
    70	# ---------------------------------------------------------------------------
    71	echo ""
    72	echo "[L1 核心层] 编译纯 C++ 核心库并执行 GoogleTest 单元测试 (零 ROS 依赖)..."
    73	if "${WORKSPACE_DIR}/scripts/build.sh"; then
    74	  record_status CORE_STATUS PASSED
    75	else
    76	  echo "[L1 核心层] FAILED：构建或单测未通过" >&2
    77	  record_status CORE_STATUS FAILED
    78	fi
    79	
    80	CORE_TEST_CASES=""
    81	CORE_TEST_SUITES=""
    82	if [ "$CORE_STATUS" = "PASSED" ]; then
    83	  CORE_TEST_CASES="$(count_ctest_cases "${WORKSPACE_DIR}/build/core")"
    84	  CORE_TEST_SUITES="$(count_suites "${WORKSPACE_DIR}/test/unit" "${WORKSPACE_DIR}/test/integration")"
    85	fi
    86	
    87	# ---------------------------------------------------------------------------
    88	# L1 压测层：基准压测（依赖 L1 构建产物）
    89	# ---------------------------------------------------------------------------
    90	BENCH_LOG="$(mktemp)"
    91	# M6：脚本早退路径也要清理 benchmark 临时文件
    92	trap 'rm -f "${BENCH_LOG}"' EXIT
    93	BENCH_QUEUE_NOTE="not measured (benchmark stage not executed)"
    94	BENCH_LOGGER_NOTE="not measured (benchmark stage not executed)"
    95	if [ "$CORE_STATUS" = "PASSED" ]; then
    96	  echo ""
    97	  echo "[L1 压测层] 执行核心库极限基准压测 (任务队列并发 & 异步低阻塞日志)..."
    98	  if "${WORKSPACE_DIR}/scripts/benchmark.sh" 2>&1 | tee "${BENCH_LOG}"; then
    99	    record_status BENCH_STATUS PASSED
   100	  else
   101	    echo "[L1 压测层] FAILED：基准压测未通过" >&2
   102	    record_status BENCH_STATUS FAILED
   103	  fi
   104	  bench_queue_push="$(sed -n 's/.*Push throughput: \([0-9.]*\) ops\/sec.*/\1/p' "${BENCH_LOG}" | head -1)"
   105	  bench_queue_pop="$(sed -n 's/.*Pop throughput:[[:space:]]*\([0-9.]*\) ops\/sec.*/\1/p' "${BENCH_LOG}" | head -1)"
   106	  bench_logger="$(sed -n 's/.*Enqueue throughput: \([0-9.]*\) ops\/sec.*/\1/p' "${BENCH_LOG}" | head -1)"
   107	  if [ "$BENCH_STATUS" = "PASSED" ]; then
   108	    if [ -n "$bench_queue_push" ] && [ -n "$bench_queue_pop" ]; then
   109	      BENCH_QUEUE_NOTE="push ${bench_queue_push} ops/sec, pop ${bench_queue_pop} ops/sec (measured this run)"
   110	    fi
   111	    if [ -n "$bench_logger" ]; then
   112	      BENCH_LOGGER_NOTE="enqueue ${bench_logger} ops/sec (measured this run)"
   113	    fi
   114	  fi
