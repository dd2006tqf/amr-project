########## launch/core_only.launch.py ##########
     1	from ament_index_python.packages import get_package_share_directory
     2	import os
     3	from launch import LaunchDescription
     4	from launch_ros.actions import LifecycleNode, Node
     5	from launch_ros.event_handlers import OnStateTransition
     6	from launch_ros.events.lifecycle import ChangeState
     7	from launch.actions import RegisterEventHandler, EmitEvent
     8	import lifecycle_msgs.msg
     9	
    10	def generate_launch_description():
    11	    # B7：与 full_system 同源的参数文件——core_only 用内联覆盖
    12	    # （容量 64 / 并发 2），仍走 yaml + 覆盖的读取路径
    13	    ros_share = get_package_share_directory('amr_dispatcher_ros')
    14	    dispatcher_yaml = os.path.join(ros_share, 'config', 'dispatcher.yaml')
    15	    stations_file_share = os.path.join(ros_share, 'config', 'stations.yaml')
    16	    safety_yaml = os.path.join(ros_share, 'config', 'safety.yaml')
    17	
    18	    dispatcher_node = LifecycleNode(
    19	        package='amr_dispatcher_ros',
    20	        executable='dispatcher_node',
    21	        name='dispatcher_node',
    22	        namespace='',
    23	        output='screen',
    24	        parameters=[
    25	            dispatcher_yaml,
    26	            {
    27	                # core_only 尺寸更小：容量/并发收紧、站点表 share 绝对路径
    28	                'queue_capacity': 64,
    29	                'max_active_missions': 2,
    30	                'stations_file': stations_file_share,
    31	            },
    32	        ]
    33	    )
    34	
    35	    # B7 补齐：core_only 之前缺 configure/activate 事件，节点永远停在 unconfigured。
    36	    # 复用 full_system 的同款模式：触发 configure，成功后 on_state_transition -> activate。
    37	    configure_event = EmitEvent(
    38	        event=ChangeState(
    39	            lifecycle_node_matcher=lambda node: node == dispatcher_node,
    40	            transition_id=lifecycle_msgs.msg.Transition.TRANSITION_CONFIGURE,
    41	        )
    42	    )
    43	
    44	    activate_event = RegisterEventHandler(
    45	        OnStateTransition(
    46	            target_lifecycle_node=dispatcher_node,
    47	            goal_state='inactive',
    48	            entities=[
    49	                EmitEvent(
    50	                    event=ChangeState(
    51	                        lifecycle_node_matcher=lambda node: node == dispatcher_node,
    52	                        transition_id=lifecycle_msgs.msg.Transition.TRANSITION_ACTIVATE,
    53	                    )
    54	                )
    55	            ],
    56	        )
    57	    )
    58	
    59	    safety_gate_node = Node(
    60	        package='amr_dispatcher_ros',
    61	        executable='safety_gate_node',
    62	        name='safety_gate_node',
    63	        output='screen',
    64	        parameters=[safety_yaml]
    65	    )
    66	
    67	    return LaunchDescription([
    68	        dispatcher_node,
    69	        configure_event,
    70	        activate_event,
    71	        safety_gate_node,
    72	    ])

   454	    # 每台车一组 path_tracker / safety_gate / chassis_driver：
   455	    #   * namespace = vehicle_id  -> 节点名唯一（/amr_01/safety_gate_node）
   456	    #   * vehicle_id 参数         -> 话题按后缀派生（/plan/amr_01、/odom/amr_01、
   457	    #                                /cmd_vel_safe/amr_01 …），绝对名不受 namespace 影响
   458	    #   * TF frame 唯一化        -> {vid}/odom、{vid}/base_link
   459	    # 兼容性红线：vehicle_ids 为空时，本函数不产生任何节点，单车路径逐字不变。
   460	    def _multi_vehicle_nodes(context):
   461	        raw_ids = launch_config_value(context, 'vehicle_ids')
   462	        ids = [t.strip() for t in raw_ids.split(',') if t.strip()]
   463	        if not ids:
   464	            return []
   465	        raw_serials = launch_config_value(context, 'serial_devices')
   466	        serials = [t.strip() for t in raw_serials.split(',') if t.strip()]
   467	        default_serial = launch_config_value(context, 'serial_device')
   468	        health = launch_config_value(context, 'health_rate_hz')
   469	        nodes = []
   470	        for idx, vid in enumerate(ids):
   471	            serial = serials[idx] if idx < len(serials) else default_serial
   472	            ns = vid
   473	            nodes.append(Node(
   474	                package='amr_dispatcher_ros',
   475	                executable='path_tracker_node',
   476	                name='path_tracker_node',
   477	                namespace=ns,
   478	                output='screen',
   479	                parameters=[{
   480	                    'vehicle_id': vid,
   481	                    'controller_type': 'pure_pursuit',
   482	                    'lookahead_distance': 0.6,
   483	                    'target_speed': 0.5,
   484	                    'control_rate_hz': 20.0,
   485	                }],
   486	                **supervision(),
   487	            ))
   488	            nodes.append(Node(
   489	                package='amr_dispatcher_ros',
   490	                executable='safety_gate_node',
   491	                name='safety_gate_node',
   492	                namespace=ns,
   493	                output='screen',
   494	                parameters=[
   495	                    safety_yaml,
   496	                    {'vehicle_id': vid},
   497	                ],
   498	                **supervision(),
   499	            ))
   500	            nodes.append(Node(
   501	                package='amr_dispatcher_ros',
   502	                executable='chassis_driver_node',
   503	                name='chassis_driver_node',
   504	                namespace=ns,
   505	                output='screen',
   506	                parameters=[
   507	                    chassis_yaml,
   508	                    {
   509	                        'vehicle_id': vid,
   510	                        'serial_device': serial,
   511	                        'health_rate_hz': ParameterValue(health, value_type=float),
   512	                    },
   513	                ],
   514	                **supervision(),
   515	            ))
   516	        return nodes
   517	
   518	    multi_vehicle_nodes = OpaqueFunction(function=_multi_vehicle_nodes)
